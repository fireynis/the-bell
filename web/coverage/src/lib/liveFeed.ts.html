import type { Post } from "../api/types";
 
/** A reaction event as it arrives over SSE. */
export interface ReactionEvent {
  post_id: string;
  reaction_type: string;
  reactor_id?: string;
}
 
/** One toast: N reactions of a given type on a single post. */
export interface ReactionNotification {
  post_id: string;
  reaction_type: string;
  count: number;
}
 
/**
 * summarizeReactions groups buffered reaction events so each notification
 * describes one post and one reaction type.
 *
 * Counting the whole buffer but labelling it with only the first event would
 * attribute reactions from several posts to whichever happened to arrive
 * first — three reactions across three posts would read as three reactions on
 * one post.
 *
 * Groups are returned in the order their first event arrived, so the oldest
 * reaction is surfaced first.
 */
export function summarizeReactions(events: readonly ReactionEvent[]): ReactionNotification[] {
  const groups = new Map<string, ReactionNotification>();
 
  for (const event of events) {
    if (!event || typeof event.post_id !== "string" || typeof event.reaction_type !== "string") {
      continue;
    }
    const key = `${event.post_id}${event.reaction_type}`;
    const existing = groups.get(key);
    if (existing) {
      existing.count++;
    } else {
      groups.set(key, {
        post_id: event.post_id,
        reaction_type: event.reaction_type,
        count: 1,
      });
    }
  }
 
  return [...groups.values()];
}
 
/**
 * mergePendingPosts prepends newly-arrived posts to the ones already waiting,
 * newest first, dropping any the reader has already seen.
 *
 * `knownIds` is re-checked at merge time rather than only on arrival: a post
 * can reach the buffer over SSE and then also arrive through normal pagination
 * before the buffer is flushed, which would otherwise show it twice.
 */
export function mergePendingPosts(
  incoming: readonly Post[],
  pending: readonly Post[],
  knownIds: ReadonlySet<string>,
): Post[] {
  const seen = new Set<string>();
  const merged: Post[] = [];
 
  for (const post of [...incoming, ...pending]) {
    if (!post || typeof post.id !== "string") continue;
    if (knownIds.has(post.id) || seen.has(post.id)) continue;
    seen.add(post.id);
    merged.push(post);
  }
 
  return merged;
}
 
const REACTION_EMOJI: Record<string, string> = {
  bell: "🔔",
  heart: "❤️",
  celebrate: "🎉",
};
 
const DEFAULT_EMOJI = "👍";
 
export function reactionEmoji(reactionType: string): string {
  return REACTION_EMOJI[reactionType] ?? DEFAULT_EMOJI;
}
 
/**
 * describeReactions composes the toast shown after a batch of reactions.
 *
 * A batch can span several posts, so the message only names a reaction type
 * when the whole batch shares one; otherwise it reports the total. Returns null
 * for an empty batch so the caller can skip showing a toast at all.
 */
export function describeReactions(notifications: readonly ReactionNotification[]): string | null {
  if (notifications.length === 0) return null;
 
  const total = notifications.reduce((sum, n) => sum + n.count, 0);
 
  if (notifications.length === 1) {
    const only = notifications[0];
    const emoji = reactionEmoji(only.reaction_type);
    return only.count === 1
      ? `Someone reacted ${emoji} to your post`
      : `${only.count} ${emoji} reactions on your post`;
  }
 
  const posts = new Set(notifications.map((n) => n.post_id)).size;
  return posts === 1
    ? `${total} reactions on your post`
    : `${total} reactions across ${posts} of your posts`;
}
 
/**
 * parseEventData safely decodes an SSE payload. Malformed JSON returns null so
 * a single bad frame cannot tear down the event listener.
 */
export function parseEventData<T>(raw: string): T | null {
  try {
    const parsed = JSON.parse(raw);
    return parsed && typeof parsed === "object" ? (parsed as T) : null;
  } catch {
    return null;
  }
}
